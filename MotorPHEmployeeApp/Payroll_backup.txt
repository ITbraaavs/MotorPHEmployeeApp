public class Payroll_backup {
    public double grossSalary;
    public double netSalary;
    public double deductions;
    
    public void calculateGross(Employee emp, Attendance att) {
        grossSalary = emp.hourlyRate * att.hoursWorked;
        System.out.println("Gross Salary: " + grossSalary);
    }
    public void calculateNet(double gross, Deduction ded)  {
        //Net salary = gross salary- total deductions
        netSalary = gross - ded.totalDeduction;
        System.out.println("Net Salary: "+ netSalary);
    }
}
